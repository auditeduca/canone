-- =============================================================================
-- 011 · TRIGGERS DE INTEGRIDADE
-- -----------------------------------------------------------------------------
-- Aqui o banco GANHA o poder de recusar. As constraints acima dizem o que é
-- válido; estas dizem o que NÃO PODE ACONTECER:
--
--   · conteúdo não publica sem verificação vigente
--   · item não publica sem exatamente uma alternativa correta
--   · norma nova VERSÃO expira a verificação de quem a citava
--   · log de revisão e log de auditoria são imutáveis
--   · projeção relacional nunca diverge do payload
--
-- POR QUE NO BANCO E NÃO SÓ NA APLICAÇÃO
-- A aplicação temfrontends, jobs, importadores e um futuro time interno.
-- Regra que mora só na aplicação é regra que alguém esquece no importador.
-- Constraint de banco não tem segunda via.
-- =============================================================================

-- -----------------------------------------------------------------------------
-- 1 · PUBLICAR EXIGE VERIFICAÇÃO VIGENTE
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_editorial.enforce_verification_on_publish() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_requires boolean;
  v_state    text;
  v_valid    int;
BEGIN
  IF NEW.state IS NOT DISTINCT FROM OLD.state THEN
    RETURN NEW;
  END IF;

  SELECT ct.requires_verification INTO v_requires
    FROM canone_editorial.content_type ct WHERE ct.id = NEW.content_type_id;

  IF NOT v_requires OR NEW.state <> 'publicado' THEN
    RETURN NEW;
  END IF;

  -- Podem existir VÁRIAS verificações para o mesmo item (tentativas
  -- recusadas, revisões). O SELECT anterior sem ORDER BY pegava uma linha
  -- ARBITRÁRIA — Bastava uma linha 'nao_verificada' mais recente para
  -- bloquear um item legítimamente verificado, ou o contrário.
  --
  -- A REGRA CORRETA: publica se EXISTE ao menos uma verificação 'verificada'
  -- e vigente. Se não existe, a mensagem mostra a decisão MAIS RECENTE.
  SELECT count(*) INTO v_valid
    FROM canone_editorial.verification v
   WHERE v.content_item_id = NEW.id
     AND v.state = 'verificada'
     AND (v.valid_until IS NULL OR v.valid_until > now());

  IF v_valid = 0 THEN
    SELECT v.state INTO v_state
      FROM canone_editorial.verification v
     WHERE v.content_item_id = NEW.id
     ORDER BY v.created_at DESC, v.id DESC
     LIMIT 1;

    RAISE EXCEPTION
      'Item % não pode ser publicado: %', NEW.key,
      CASE
        WHEN v_state IS NULL THEN 'verificação ausente'
        WHEN v_state = 'expirada' THEN 'a verificação expirou'
        ELSE 'verificação em estado ' || v_state
      END
      USING ERRCODE = '23514',
            HINT = 'Passe por canone_editorial.verification com state=''verificada'' e trecho da fonte.';
  END IF;

  NEW.published_at := now();
  RETURN NEW;
END $$;

CREATE TRIGGER trg_publish_requires_verification
  BEFORE UPDATE OF state ON canone_editorial.content_item
  FOR EACH ROW EXECUTE FUNCTION canone_editorial.enforce_verification_on_publish();

-- -----------------------------------------------------------------------------
-- 2 · NORMA NOVA VERSÃO EXPIRA A VERIFICAÇÃO DE QUEM A CITAVA
-- -----------------------------------------------------------------------------
-- Este é o trigger que mantém a promessa do produto honesta. Uma norma muda
-- → quem citava precisa rever → a fila de moderação abre sozinha.
-- Ninguém precisa lembrar de rodar um relatório.
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_norms.expire_verifications_on_new_version() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_norm_code text;
  v_count     int;
BEGIN
  SELECT n.code INTO v_norm_code
    FROM canone_norms.norm_version nv
    JOIN canone_norms.norm n ON n.id = nv.norm_id
   WHERE nv.id = NEW.id;

  -- Itens que CITAM esta norma. A aresta é canone_editorial.content_norm_citation
  -- (a versão anterior fazia JOIN com canone_norms.norm_item por um id que era
  -- content_item — e não casava nunca).
  WITH citadas AS (
    SELECT DISTINCT c.content_item_id AS id
      FROM canone_editorial.content_norm_citation c
     WHERE c.norm_id = (SELECT id FROM canone_norms.norm WHERE code = v_norm_code)
       AND c.norm_version_id <> NEW.id
  )
  UPDATE canone_editorial.verification v
     SET state = 'necessita_atualizacao',
         updated_at = now()
   WHERE v.content_item_id IN (SELECT id FROM citadas)
     AND v.state IN ('verificada','nao_verificada','em_revisao');

  GET DIAGNOSTICS v_count = ROW_COUNT;

  IF v_count > 0 THEN
    INSERT INTO canone_admin.moderation_queue (content_item_id, queue, priority, reason, detail)
    SELECT c.id, 'expirada', 80,
           'Norma ' || v_norm_code || ' teve nova versão (' || NEW.label || ').',
           jsonb_build_object('norm_code', v_norm_code, 'new_version', NEW.label,
                              'effective_from', NEW.effective_from)
      FROM (
        SELECT DISTINCT c.content_item_id AS id
          FROM canone_editorial.content_norm_citation c
         WHERE c.norm_id = (SELECT id FROM canone_norms.norm WHERE code = v_norm_code)
           AND c.norm_version_id <> NEW.id
      ) c
      WHERE NOT EXISTS (
        SELECT 1 FROM canone_admin.moderation_queue q
         WHERE q.content_item_id = c.id AND q.queue = 'expirada' AND q.resolved_at IS NULL
      );
  END IF;

  RETURN NEW;
END $$;

CREATE TRIGGER trg_new_norm_version_expires
  AFTER INSERT ON canone_norms.norm_version
  FOR EACH ROW EXECUTE FUNCTION canone_norms.expire_verifications_on_new_version();

-- -----------------------------------------------------------------------------
-- 3 · ITEM DE AVALIAÇÃO EXIGE EXATAMENTE UMA ALTERNATIVA CORRETA
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_learning.enforce_single_correct_option() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_count int;
BEGIN
  IF NEW.is_correct THEN
    SELECT count(*) INTO v_count FROM canone_learning.item_option
      WHERE item_id = NEW.item_id AND is_correct;
    IF v_count > 1 THEN
      RAISE EXCEPTION 'Item % teria % alternativas corretas; o edital admite uma',
        NEW.item_id, v_count USING ERRCODE = '23514';
    END IF;
  END IF;
  RETURN NEW;
END $$;

CREATE TRIGGER trg_single_correct_option
  BEFORE INSERT OR UPDATE OF is_correct ON canone_learning.item_option
  FOR EACH ROW EXECUTE FUNCTION canone_learning.enforce_single_correct_option();

-- E o item não publica sem nenhuma alternativa correta.
CREATE OR REPLACE FUNCTION canone_learning.require_correct_option_to_publish() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_correct  int;
  v_options  int;
  v_item_id  uuid;
  v_family   text;
BEGIN
  IF NEW.state <> 'publicado' OR OLD.state = 'publicado' THEN
    RETURN NEW;
  END IF;

  -- A REGRA É DO TIPO, não da projeção. Se o item é de AVALIAÇÃO, ele tem
  -- exatamente uma alternativa correta — mesmo que ainda não tenha NENHUMA
  -- versão de payload. A versão anterior devolvia cedo quando não havia
  -- projeção ('IF v_item_id IS NULL THEN RETURN NEW'), e por isso um item
  -- de avaliação sem payload era publicado com ZERO alternativas.
  SELECT ct.family INTO v_family
    FROM canone_editorial.content_type ct
   WHERE ct.id = NEW.content_type_id;

  IF v_family <> 'avaliacao' THEN
    RETURN NEW;  -- conteúdo de estudo: a regra não se aplica
  END IF;

  -- A trigger vive em canone_editorial.content_item; as alternativas vivem em
  -- canone_learning.item_option, que referencia canone_learning.item.id — NÃO o id do
  -- content_item. Sem esta ponte a contagem daria 0 e TODO item seria
  -- publicado sem alternativa correta.
  SELECT i.id INTO v_item_id
    FROM canone_learning.item i
   WHERE i.content_item_id = NEW.id;

  IF v_item_id IS NULL THEN
    RAISE EXCEPTION
      'Item % não pode ser publicado: item de avaliação sem alternativas (exige exatamente 1)',
      NEW.key
      USING ERRCODE = '23514';
  END IF;

  SELECT count(*) FILTER (WHERE is_correct), count(*)
    INTO v_correct, v_options
    FROM canone_learning.item_option WHERE item_id = v_item_id;

  IF v_options = 0 OR v_correct <> 1 THEN
    RAISE EXCEPTION
      'Item % não pode ser publicado: % alternativas, % corretas; exige exatamente 1',
      NEW.key, v_options, v_correct
      USING ERRCODE = '23514';
  END IF;
  RETURN NEW;
END $$;

CREATE TRIGGER trg_require_correct_option
  BEFORE UPDATE OF state ON canone_editorial.content_item
  FOR EACH ROW EXECUTE FUNCTION canone_learning.require_correct_option_to_publish();

-- -----------------------------------------------------------------------------
-- 4 · PROJEÇÃO RELACIONAL DERIVADA DO PAYLOAD (sem dual-write manual)
-- -----------------------------------------------------------------------------
-- O payload é canônico. A projeção é recalculada por trigger. Duas coisas
-- que isso compra:
--   · nunca há divergência entre o que o editor vê e o que o banco consulta
--   · a regra "exatamente uma alternativa correta" é verificável no payload
--     E na projeção — porque a projeção É o payload projetado
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_editorial.sync_learning_projection() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  -- DOIS identificadores, e confundi-los era o bug:
  --   v_content_item = canone_editorial.content_item.id  (o documento)
  --   v_item         = canone_learning.item.id           (a PROJEÇÃO)
  -- canone_learning.item_option.item_id referencia canone_learning.item.id, NÃO o
  -- content_item. A versão anterior gravava v_item (= content_item.id) e
  -- violava a FK em toda criação de item.
  v_content_item uuid;
  v_item         uuid;
  v_kind text;
  v_payload jsonb;
  v_opts  jsonb;
  v_correct_count int;
BEGIN
  IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;

  v_content_item := NEW.content_item_id;
  v_payload := NEW.payload;
  v_kind := coalesce(v_payload->>'kind', 'questao');

  IF v_kind NOT IN ('questao','flashcard','caso','simulado_item','calculo') THEN
    RETURN NEW;  -- não é item de avaliação; nada a projetar
  END IF;

  INSERT INTO canone_learning.item (content_item_id, content_type_key, kind, difficulty, estimated_seconds)
  VALUES (v_content_item, (SELECT ct.key FROM canone_editorial.content_type ct WHERE ct.id =
                    (SELECT content_type_id FROM canone_editorial.content_item WHERE id = v_content_item)),
          v_kind,
          coalesce((v_payload->>'difficulty')::numeric, 2.00),
          (v_payload->>'estimated_seconds')::int)
  ON CONFLICT (content_item_id) DO UPDATE
    SET kind = EXCLUDED.kind,
        difficulty = EXCLUDED.difficulty,
        estimated_seconds = EXCLUDED.estimated_seconds,
        updated_at = now();

  -- O id da PROJEÇÃO (não do documento).
  SELECT i.id INTO v_item
    FROM canone_learning.item i
   WHERE i.content_item_id = v_content_item;

  -- Alternativas: substituem o conjunto inteiro (o payload é a verdade).
  DELETE FROM canone_learning.item_option WHERE item_id = v_item;

  v_opts := v_payload->'options';
  IF v_opts IS NOT NULL AND jsonb_typeof(v_opts) = 'array' THEN
    -- jsonb_array_elements(...) WITH ORDINALITY devolve DUAS colunas:
    -- (value, n_ord). A versão anterior (AS opt(n, n_ord)) ligava o nome
    -- 'opt' à COLUNA n — e 'opt->>x' sobre bigint é erro de tipo. A
    -- PROJEÇÃO NUNCA FUNCIONOU: todo item com alternativas derrubava no
    -- trigger. Foi o teste de comportamento que pegou — a migration
    -- aplicava sem erro, o que não é prova de que a regra funciona.
    INSERT INTO canone_learning.item_option (item_id, ordinal, body, is_correct, distractor_kind, ordinal_pos)
    SELECT v_item,
           upper(coalesce(opt.value->>'ordinal', chr((opt.n_ord - 1)::int + 65)::text)),
           coalesce(opt.value->>'body',''),
           coalesce((opt.value->>'correct')::boolean, false),
           opt.value->>'distractor_kind',
           opt.n_ord
      FROM jsonb_array_elements(v_opts) WITH ORDINALITY AS opt(value, n_ord)
      WHERE jsonb_typeof(opt.value) = 'object';

    SELECT count(*) FILTER (WHERE is_correct) INTO v_correct_count
      FROM canone_learning.item_option WHERE item_id = v_item;

    IF v_correct_count > 1 THEN
      RAISE EXCEPTION
        'Payload do item % declara % alternativas corretas; o edital admite uma',
        v_content_item, v_correct_count USING ERRCODE = '23514';
    END IF;
  END IF;

  RETURN NEW;
END $$;

CREATE TRIGGER trg_sync_learning_projection
  AFTER INSERT OR UPDATE OF payload ON canone_editorial.content_item_version
  FOR EACH ROW EXECUTE FUNCTION canone_editorial.sync_learning_projection();

-- -----------------------------------------------------------------------------
-- 5 · TRANSIÇÃO DE ESTADO VÁLIDA
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_editorial.enforce_state_transition() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_from      text;
  v_to        text;
  v_family    text;
  v_requires  text;
  v_allow     boolean;
BEGIN
  IF NEW.state IS NOT DISTINCT FROM OLD.state THEN RETURN NEW; END IF;

  -- A família vem do TIPO, pela versão de tipo vinculada ao item.
  -- (A versão anterior fazia JOIN por ctv.id = ct.id — chaves de tabelas
  --  diferentes. O JOIN não filtrava nada e podia devolver a família
  --  errada.)
  SELECT ct.family INTO v_family
    FROM canone_editorial.content_type ct
   WHERE ct.id = NEW.content_type_id;

  -- Estados fora do catálogo são erro de digitação, não transição.
  IF NOT EXISTS (SELECT 1 FROM canone_editorial.state_definition WHERE family = v_family AND key = NEW.state) THEN
    RAISE EXCEPTION 'Estado % inexistente para a família %', NEW.state, v_family
      USING ERRCODE = '23514';
  END IF;

  v_from := OLD.state; v_to := NEW.state;

  -- A TRANSIÇÃO PRECISA ESTAR NA TABELA. Caminho não catalogado é
  -- recusado, não aceito por omissão.
  SELECT requires INTO v_requires
    FROM canone_editorial.state_transition
   WHERE family = v_family AND from_state = v_from AND to_state = v_to;

  v_allow := (v_requires IS NOT NULL) OR (v_from IS NULL);

  IF NOT v_allow THEN
    RAISE EXCEPTION
      'Transição não permitida: % → % (família %) não está em canone_editorial.state_transition',
      v_from, v_to, v_family
      USING ERRCODE = '23514';
  END IF;

  -- Transição que exige verificação vigente só passa se a houver.
  IF v_requires = 'verificacao_vigente' THEN
    IF NOT EXISTS (
      SELECT 1
        FROM canone_editorial.verification v
       WHERE v.content_item_id = NEW.id
         AND v.state = 'verificada'
         AND (v.valid_until IS NULL OR v.valid_until > now())
    ) THEN
      RAISE EXCEPTION
        'Item % não pode ir para % sem verificação de fonte vigente',
        NEW.key, v_to
        USING ERRCODE = '23514';
    END IF;
  END IF;

  -- Grava o histórico de workflow, sempre.
  INSERT INTO canone_editorial.content_revision
    (content_item_id, version_id, from_state, to_state, actor_id, reason)
  VALUES (NEW.id, NEW.current_version_id, v_from, v_to, canone_ops.current_user_id(), NULL);

  RETURN NEW;
END $$;

CREATE TRIGGER trg_state_transition
  BEFORE UPDATE OF state ON canone_editorial.content_item
  FOR EACH ROW EXECUTE FUNCTION canone_editorial.enforce_state_transition();

-- -----------------------------------------------------------------------------
-- 6 · updated_at automático
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_ops.touch_updated_at() RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  NEW.updated_at := now();
  RETURN NEW;
END $$;

DO $$
DECLARE
  t     text;
  v_sch text;
  v_tab text;
BEGIN
  FOREACH t IN ARRAY ARRAY[
    'canone_identity.account','canone_identity.user_profile','canone_editorial.content_item','canone_editorial.verification',
    'canone_learning.item','canone_learning.user_item_state','canone_learning.mastery','canone_learning.note',
    'canone_commerce.plan','canone_commerce.subscription','canone_ops.setting','canone_ops.integration','canone_ops.webhook'
  ] LOOP
    v_sch := split_part(t, '.', 1);
    v_tab := split_part(t, '.', 2);
    -- %I.%I: sem e vírgula o %I citaria "canone_identity.account" como UM nome só.
    EXECUTE format(
      'CREATE TRIGGER trg_touch_%s BEFORE UPDATE ON %I.%I FOR EACH ROW EXECUTE FUNCTION canone_ops.touch_updated_at()',
      replace(t,'.','_'), v_sch, v_tab);
  END LOOP;
END $$;

-- -----------------------------------------------------------------------------
-- 7 · CONCESSÃO DE VISÃO DE USUÁRIO EXPIRA SOZINHA
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_admin.expire_support_grants() RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  -- TRAVA DE REENTRÂNCIA. O UPDATE interno dispara este MESMO trigger
  -- (AFTER UPDATE, FOR EACH STATEMENT). Um trigger de statement roda uma
  -- vez por statement — mesmo quando afeta ZERO linhas — e a função
  -- sempre emite um UPDATE. Sem a trava: UPDATE → trigger → UPDATE →
  -- trigger → ... até o stack estourar. Foi o que aconteceu; o teste
  -- crashed com "memory access out of bounds" (estouro de pilha no WASM).
  IF current_setting('app.expiring_grants', true) = '1' THEN
    RETURN NULL;
  END IF;

  PERFORM set_config('app.expiring_grants', '1', true);

  UPDATE canone_admin.support_access_grant
     SET revoked_at = now(), revoke_reason = 'expirada automaticamente'
   WHERE revoked_at IS NULL AND expires_at <= now();

  PERFORM set_config('app.expiring_grants', '0', true);
  RETURN NULL;
END $$;

-- A expiração é AUTOMÁTICA, não depende de cron. A concessão morre no
-- instante em que expira — mesmo que ninguém rode o job de limpeza.
-- Statement-level: revarre o conjunto, não a linha.
CREATE TRIGGER trg_expire_support_grants
  AFTER INSERT OR UPDATE ON canone_admin.support_access_grant
  FOR EACH STATEMENT EXECUTE FUNCTION canone_admin.expire_support_grants();

-- Dispara a cada minuto, via pg_cron no ambiente de produção.
-- (Em PGlite/sandbox, o teste chama a função diretamente.)
CREATE OR REPLACE FUNCTION canone_admin.purge_expired_grants() RETURNS int
LANGUAGE plpgsql AS $$
DECLARE v_count int;
BEGIN
  UPDATE canone_admin.support_access_grant
     SET revoked_at = now(), revoke_reason = 'expirada automaticamente'
   WHERE revoked_at IS NULL AND expires_at <= now();
  GET DIAGNOSTICS v_count = ROW_COUNT;
  RETURN v_count;
END $$;
