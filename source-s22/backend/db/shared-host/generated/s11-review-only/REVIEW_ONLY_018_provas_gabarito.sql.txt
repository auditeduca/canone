-- POR QUE ESTA MIGRATION EXISTE
--
-- O schema até aqui tratava o GABARITO como um atributo do item (uma coluna

-- coisas estão erradas para o acervo real:

--    gabarito oficial e não é "a prova de 2024.2".
--
-- 2. O gabarito oficial NÃO é um booleano. Ele tem CICLO DE VIDA
--    (preliminar → definitivo), tem ANULAÇÃO (questão anulada não tem

--    exige EXATAMENTE UMA alternativa correta, e uma questão anulada não
--    tem gabarito nenhum. Sem esta migration, o acervo real (34 questões
--    anuladas) simplesmente não entraria.

--
--   taxonomy.exam_edition   a RODADA (2025.1, 2026.2 …)
--     └─ taxonomy.exam_track   o EXAME dentro da rodada

-- -----------------------------------------------------------------------------
-- 5 · O GABARITO VIGENTE (precedência)
-- -----------------------------------------------------------------------------
-- Qual gabarito manda para um item: o DEFINITIVO se existir, senão o
-- preliminar. Esta é a única fonte de veredito do servidor.
CREATE OR REPLACE FUNCTION canone_learning.vigente_answer_key(p_item_id uuid)
RETURNS TABLE (
  answer_key_id uuid, key_status text, ordinal int,
  correct_option text, is_annulled boolean
)
LANGUAGE sql STABLE AS $$
  SELECT ak.id, ak.status, e.ordinal, e.correct_option, e.is_annulled
    FROM learning.exam_form_item efi
    JOIN learning.exam_form ef  ON ef.id = efi.exam_form_id
    JOIN learning.answer_key ak ON ak.exam_form_id = ef.id
    JOIN learning.answer_key_entry e
      ON e.answer_key_id = ak.id AND e.ordinal = efi.ordinal
   WHERE efi.item_id = p_item_id
   ORDER BY (ak.status = 'definitivo') DESC,
            ak.published_at DESC NULLS LAST,
            ak.created_at DESC
   LIMIT 1
$$;
COMMENT ON FUNCTION canone_learning.vigente_answer_key(uuid) IS
  'Gabarito vigente de um item: o definitivo, senão o preliminar. NULL se não houver.';

-- -----------------------------------------------------------------------------
-- 6 · PROJEÇÃO: O GABARITO ESCREVE A LETRA NO ITEM
-- -----------------------------------------------------------------------------
-- O gabarito oficial tem PRECEDÊNCIA sobre a letra que veio no payload.
-- Por quê: o payload é o que o Cânone escreveu; o gabarito é o que o CFC
-- publicou. Quando os dois divergem, quem está certo é o CFC — e a
-- divergência tem que ser resolvida, não silenciada.
CREATE OR REPLACE FUNCTION canone_learning.set_item_correct(p_item_id uuid, p_option text)
RETURNS void
LANGUAGE plpgsql AS $$
BEGIN
  IF p_option IS NULL THEN RETURN; END IF;
  -- Duas instruções, nesta ordem: na primeira o trigger de "no máximo uma
  -- correta" vê zero corretas; na segunda ele vê zero também. Numa
  -- instrução só, o trigger contaria a alternativa antiga e a nova juntas.
  UPDATE learning.item_option SET is_correct = false
   WHERE item_id = p_item_id AND is_correct;
  UPDATE learning.item_option SET is_correct = true
   WHERE item_id = p_item_id AND ordinal = p_option;
END $$;

-- Reaplica o gabarito vigente a um item (usado quando o payload muda).
CREATE OR REPLACE FUNCTION canone_learning.apply_answer_key(p_item_id uuid)
RETURNS void
LANGUAGE plpgsql AS $$
DECLARE
  v_entry record;
BEGIN
  SELECT * INTO v_entry FROM learning.vigente_answer_key(p_item_id);
  IF NOT FOUND THEN RETURN; END IF;

  IF v_entry.is_annulled THEN
    PERFORM learning.clear_item_correct(p_item_id);
    RETURN;
  END IF;

  PERFORM learning.set_item_correct(p_item_id, v_entry.correct_option);
END $$;

-- QUESTÃO ANULADA NÃO TEM RESPOSTA. O CFC anulou porque a questão é
-- defeituosa (ambígua, com negação dupla, com gabarito errado): manter
-- uma "alternativa correta" seria affirmar que a questão tem resposta
-- quando o órgão máximo da profissão disse que não tem.
--
-- E ANULAR UMA QUESTÃO JÁ PUBLICADA É RECUSADO: quem praticou aquela
-- questão precisa ver a retração happen. Silenciar isso seria trocar a
-- nota de um aluno sem ele saber.
CREATE OR REPLACE FUNCTION canone_learning.clear_item_correct(p_item_id uuid)
RETURNS void
LANGUAGE plpgsql AS $$
DECLARE
  v_state text;
BEGIN
  SELECT ci.state INTO v_state
    FROM learning.item i
    JOIN editorial.content_item ci ON ci.id = i.content_item_id
   WHERE i.id = p_item_id;

  IF v_state = 'publicado' THEN
    RAISE EXCEPTION
      'A questão % está publicada: reverta a publicação antes de anular', p_item_id
      USING ERRCODE = '23514';
  END IF;

  UPDATE learning.item_option SET is_correct = false
   WHERE item_id = p_item_id AND is_correct;
END $$;

-- -----------------------------------------------------------------------------
-- 7 · VALIDAÇÃO DA ENTRADA DO GABARITO (imediata)
-- -----------------------------------------------------------------------------
-- Três recusas, todas as que a extração de gabarito pode errar:
--   a) a posição não existe na prova;
--   b) a prova não tem item nessa posição;
--   c) a letra do gabarito não existe entre as alternativas da questão.
-- A (c) é o detector de PROVA DESALINHADA — exatamente o defeito que
-- corrompe um acervo inteiro quando o parser perde uma seção do PDF.
CREATE OR REPLACE FUNCTION canone_learning.check_answer_key_entry() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_key      learning.answer_key%ROWTYPE;
  v_form_q   int;
  v_item     uuid;
  v_has      int;
BEGIN
  SELECT * INTO v_key FROM learning.answer_key WHERE id = NEW.answer_key_id;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Gabarito % inexistente', NEW.answer_key_id USING ERRCODE = '23503';
  END IF;

  SELECT t.question_count INTO v_form_q
    FROM learning.exam_form ef
    JOIN taxonomy.exam_track t ON t.id = ef.exam_track_id
   WHERE ef.id = v_key.exam_form_id;

  IF NEW.ordinal > v_form_q THEN
    RAISE EXCEPTION 'Posição % não existe na prova (que tem % questões)',
      NEW.ordinal, v_form_q USING ERRCODE = '23514';
  END IF;

  SELECT efi.item_id INTO v_item
    FROM learning.exam_form_item efi
   WHERE efi.exam_form_id = v_key.exam_form_id
     AND efi.ordinal = NEW.ordinal;
  IF v_item IS NULL THEN
    RAISE EXCEPTION 'A prova % não tem item na posição %',
      v_key.exam_form_id, NEW.ordinal USING ERRCODE = '23514';
  END IF;

  IF NOT NEW.is_annulled THEN
    SELECT count(*) INTO v_has
      FROM learning.item_option
     WHERE item_id = v_item AND ordinal = NEW.correct_option;
    IF v_has = 0 THEN
      RAISE EXCEPTION
        'Gabarito aponta para a alternativa % que não existe na questão da posição %',
        NEW.correct_option, NEW.ordinal USING ERRCODE = '23514';
    END IF;
  END IF;

  RETURN NEW;
END $$;

CREATE TRIGGER trg_check_answer_key_entry
  BEFORE INSERT OR UPDATE ON canone_learning.answer_key_entry
  FOR EACH ROW EXECUTE FUNCTION canone_learning.check_answer_key_entry();

-- -----------------------------------------------------------------------------
-- 9 · O GABARITO DEFINITIVO TEM QUE ESTAR COMPLETO
-- -----------------------------------------------------------------------------
-- CONSTRAINT TRIGGER DEFERRABLE: a verificação acontece no COMMIT, quando
-- as entradas já foram inseridas. Sem isso, a regra só valeria na entrada
-- — e "regra que só vale na entrada" é regra morta.
--
-- Esta é a regra que impede o acervo com 39 posições erradas: se o parser
-- perdeu uma seção do PDF, a contagem não bate e a transação INTEIRA cai.
CREATE OR REPLACE FUNCTION canone_learning.assert_answer_key_complete(p_key_id uuid)
RETURNS void
LANGUAGE plpgsql AS $$
DECLARE
  v_status   text;
  v_form_id  uuid;
  v_expected int;
  v_linked   int;
  v_count    int;
  v_min      int;
  v_max      int;
BEGIN
  SELECT ak.status, ak.exam_form_id INTO v_status, v_form_id
    FROM learning.answer_key ak WHERE ak.id = p_key_id;
  IF v_status IS DISTINCT FROM 'definitivo' THEN RETURN; END IF;

  SELECT t.question_count INTO v_expected
    FROM learning.exam_form ef
    JOIN taxonomy.exam_track t ON t.id = ef.exam_track_id
   WHERE ef.id = v_form_id;

  -- REGRA A: a prova tem que estar INTEIRA. Um gabarito definitivo sobre
  -- uma prova pela metade é o jeito mais elegante de perder questões em
  -- silêncio — e a perda é justamente o que o acervo não pode ter.
  SELECT count(*) INTO v_linked
    FROM learning.exam_form_item efi
   WHERE efi.exam_form_id = v_form_id;

  IF v_linked <> v_expected THEN
    RAISE EXCEPTION
      'A prova % tem % questões vinculadas de %; gabarito definitivo exige a prova completa',
      v_form_id, v_linked, v_expected USING ERRCODE = '23514';
  END IF;

  -- REGRA B: o gabarito tem que cobrir TODAS as posições, sem buraco.
  SELECT count(*), min(e.ordinal), max(e.ordinal)
    INTO v_count, v_min, v_max
    FROM learning.answer_key_entry e
   WHERE e.answer_key_id = p_key_id;

  IF v_count <> v_expected OR v_min IS DISTINCT FROM 1 OR v_max IS DISTINCT FROM v_expected THEN
    RAISE EXCEPTION
      'Gabarito % tem % entradas (posições %..%); a prova tem % questões. Gabarito definitivo exige todas.',
      p_key_id, v_count, coalesce(v_min, 0), coalesce(v_max, 0), v_expected
      USING ERRCODE = '23514';
  END IF;
END $$;

-- O arguments de CREATE TRIGGER são CONSTANTES DE STRING: não se pode
-- passar NEW.id. A função de trigger lê a própria linha — por isso os
-- wrappers abaixo, em vez de "EXECUTE FUNCTION f(NEW.id)" (que o Postgres
-- rejeita com "syntax error at or near .").
CREATE OR REPLACE FUNCTION canone_learning.trg_assert_answer_key_complete() RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  PERFORM learning.assert_answer_key_complete(NEW.id);
  RETURN NULL;
END $$;

-- -----------------------------------------------------------------------------
-- 11 · A TRILHA DA SUFICIÊNCIA TEM DE BATER COM A RODADA
-- -----------------------------------------------------------------------------
-- A rodada tem 50 questões e aprovação 25. A trilha da Suficência tem que
-- dizer o mesmo — senão o mesmo exame tem dois números verdadeiros.
CREATE OR REPLACE FUNCTION canone_learning.check_exam_track_numbers() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_ed_q     int;
  v_ed_score int;
BEGIN
  IF NEW.track_code <> 'suficiencia' THEN RETURN NEW; END IF;

  SELECT e.question_count, e.passing_score
    INTO v_ed_q, v_ed_score
    FROM taxonomy.exam_edition e
   WHERE e.id = NEW.exam_edition_id;

  IF v_ed_q IS NOT NULL AND v_ed_q <> NEW.question_count THEN
    RAISE EXCEPTION
      'Trilha de Suficiência com % questões; a rodada % tem %',
      NEW.question_count, NEW.exam_edition_id, v_ed_q USING ERRCODE = '23514';
  END IF;
  IF v_ed_score IS NOT NULL AND NEW.passing_score IS NOT NULL AND v_ed_score <> NEW.passing_score THEN
    RAISE EXCEPTION
      'Trilha de Suficência com aprovação %; a rodada % tem %',
      NEW.passing_score, NEW.exam_edition_id, v_ed_score USING ERRCODE = '23514';
  END IF;
  RETURN NEW;
END $$;

CREATE TRIGGER trg_exam_track_numbers
  BEFORE INSERT OR UPDATE ON canone_taxonomy.exam_track
  FOR EACH ROW EXECUTE FUNCTION canone_learning.check_exam_track_numbers();

-- -----------------------------------------------------------------------------
-- 12 · A CORREÇÃO PASSA A DISTINGUIR ANULADA
-- -----------------------------------------------------------------------------
-- Antes: boolean. Agora: 'correta' | 'errada' | 'anulada'.
-- Uma questão anulada pelo CFC NÃO é errada — o candidato não tem como
-- errá-la. Devolver 'errada' seria mentir sobre o próprio exame.
DROP FUNCTION IF EXISTS canone_learning.grade_answer(uuid, text);

CREATE OR REPLACE FUNCTION canone_learning.grade_answer(p_item_id uuid, p_ordinal text)
RETURNS text
LANGUAGE plpgsql
SECURITY DEFINER
STABLE AS $$
DECLARE
  v_entry record;
  v_ok    boolean;
BEGIN
  SELECT * INTO v_entry FROM learning.vigente_answer_key(p_item_id);
  IF FOUND THEN
    IF v_entry.is_annulled THEN
      RETURN 'anulada';
    END IF;
    RETURN CASE WHEN upper(p_ordinal) = v_entry.correct_option THEN 'correta' ELSE 'errada' END;
  END IF;

  -- Sem gabarito oficial: vale o gabarito do próprio item.
  SELECT EXISTS (
    SELECT 1 FROM learning.item_option
     WHERE item_id = p_item_id AND ordinal = upper(p_ordinal) AND is_correct
  ) INTO v_ok;
  RETURN CASE WHEN v_ok THEN 'correta' ELSE 'errada' END;
END $$;
COMMENT ON FUNCTION canone_learning.grade_answer(uuid, text) IS
  'Corrige a resposta. Devolve o VEREDITO: correta | errada | anulada. O gabarito não sai do servidor.';

-- -----------------------------------------------------------------------------
-- 13 · O PAYLOAD NÃO VOLTA A SOBRESCREVER O GABARITO
-- -----------------------------------------------------------------------------
-- A projeção de learning.* é reescrita a cada mudança de payload (o payload
-- é canônico). Sem esta correção, editar o enunciado de uma questão de
-- prova DEVOLVERIA a letra antiga do payload e o gabarito oficial
-- silenciosamente deixaria de valer. O gabarito tem precedência — sempre.
CREATE OR REPLACE FUNCTION canone_editorial.sync_learning_projection() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_content_item uuid;
  v_item         uuid;
  v_kind text;
  v_payload jsonb;
  v_opts  jsonb;
  v_correct_count int;

BEGIN
  IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;

  v_content_item := NEW.content_item_id;
  v_payload := NEW.payload;
  v_kind := coalesce(v_payload->>'kind', 'questao');

  IF v_kind NOT IN ('questao','flashcard','caso','simulado_item','calculo') THEN
    RETURN NEW;  -- não é item de avaliação; nada a projetar
  END IF;

  INSERT INTO learning.item (content_item_id, content_type_key, kind, difficulty, estimated_seconds)
  VALUES (v_content_item, (SELECT ct.key FROM editorial.content_type ct WHERE ct.id =
                    (SELECT content_type_id FROM editorial.content_item WHERE id = v_content_item)),
          v_kind,
          coalesce((v_payload->>'difficulty')::numeric, 2.00),
          (v_payload->>'estimated_seconds')::int)
  ON CONFLICT (content_item_id) DO UPDATE
    SET kind = EXCLUDED.kind,
        difficulty = EXCLUDED.difficulty,
        estimated_seconds = EXCLUDED.estimated_seconds,
        updated_at = now();

  SELECT i.id INTO v_item
    FROM learning.item i
   WHERE i.content_item_id = v_content_item;

  DELETE FROM learning.item_option WHERE item_id = v_item;

  v_opts := v_payload->'options';
  IF v_opts IS NOT NULL AND jsonb_typeof(v_opts) = 'array' THEN
    INSERT INTO learning.item_option (item_id, ordinal, body, is_correct, distractor_kind, ordinal_pos)
    SELECT v_item,
           upper(coalesce(opt.value->>'ordinal', chr((opt.n_ord - 1)::int + 65)::text)),
           coalesce(opt.value->>'body',''),
           coalesce((opt.value->>'correct')::boolean, false),
           opt.value->>'distractor_kind',
           opt.n_ord
      FROM jsonb_array_elements(v_opts) WITH ORDINALITY AS opt(value, n_ord)
     WHERE jsonb_typeof(opt.value) = 'object';

    SELECT count(*) FILTER (WHERE is_correct) INTO v_correct_count
      FROM learning.item_option WHERE item_id = v_item;

    IF v_correct_count > 1 THEN
      RAISE EXCEPTION
        'Payload do item % declara % alternativas corretas; o edital admite uma',
        v_content_item, v_correct_count USING ERRCODE = '23514';
    END IF;
  END IF;

  -- O GABARITO OFICIAL VENCE O PAYLOAD. A letra publicada no item é a que
  -- o CFC imprimiu — não a que o redator deixou no enunciado.
  PERFORM learning.apply_answer_key(v_item);

  RETURN NEW;
END $$;

-- 15 · VISÃO DE GOVERNANÇA (admin)
-- -----------------------------------------------------------------------------
CREATE OR REPLACE VIEW canone_admin.v_exam_form_status AS
SELECT ef.id                AS exam_form_id,
       ef.form_code,
       ef.form_label,
       e.code               AS edition_code,
       e.status             AS edition_status,
       t.track_code,
       t.specialty_code,
       t.question_count,
       (SELECT count(*) FROM canone_learning.exam_form_item efi WHERE efi.exam_form_id = ef.id) AS items_linked,
       (SELECT ak.status FROM canone_learning.answer_key ak WHERE ak.exam_form_id = ef.id ORDER BY (ak.status='definitivo') DESC LIMIT 1) AS key_status,
       (SELECT count(*) FROM canone_learning.answer_key_entry e2
          JOIN canone_learning.answer_key ak2 ON ak2.id = e2.answer_key_id
         WHERE ak2.exam_form_id = ef.id AND ak2.status = 'definitivo' AND e2.is_annulled) AS annulled,
       (SELECT count(*) FROM canone_learning.answer_key_entry e3
          JOIN canone_learning.answer_key ak3 ON ak3.id = e3.answer_key_id
         WHERE ak3.exam_form_id = ef.id AND ak3.status = 'definitivo') AS keyed
  FROM canone_learning.exam_form ef
  JOIN canone_taxonomy.exam_track t ON t.id = ef.exam_track_id
  JOIN canone_taxonomy.exam_edition e ON e.id = t.exam_edition_id;

COMMENT ON VIEW canone_admin.v_exam_form_status IS
  'Situação de cada variante de prova: questões vinculadas, gabarito vigente, anuladas.';
