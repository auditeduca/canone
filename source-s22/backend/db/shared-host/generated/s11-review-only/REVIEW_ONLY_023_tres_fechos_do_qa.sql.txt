-- FURO 1 · A ÁREA ERA DECIDIDA POR UM GUC QUE QUALQUER UM ESCREVE
--
-- ops.is_staff() lia o GUC `app.current_area`. GUC de texto livre é
-- escrevível por QUALQUER sessão: `SET app.current_area = 'staff'` funciona
-- como assinante. E o QA mediu a consequência: com a área forjada, o
-- assinante REVELAVA O GABARITO ANTES DE TERMINAR A TENTATIVA
-- (o "gabarito só é revelado depois de terminar" era decorativo) e ainda
-- FINALIZAVA a tentativa alheia.
--
-- A CORREÇÃO: a área passa a ser decidida pelo PAPEL (current_user), que
-- só muda por SET ROLE — e SET ROLE exige MEMBRO do papel de destino. O
-- assinante não é membro de canone_staff (verificado em pg_auth_members:
-- só canone_app é, com NOINHERIT).
--
-- Detalhe que importa: dentro de uma função SECURITY DEFINER, current_user
-- é o DONO da função. Por isso a fonte é o GUC `role` (que o SET ROLE
-- escreve e que ninguém consegue forjar sem ser membro) com current_user
-- como reserva. `app.current_area` continua existindo — como informação de
-- DIAGNÓSTICO, nunca como autorização.

-- FURO 2 · O INSERT PODIA CRIAR A SEGUNDA ALTERNATIVA CORRETA
--
-- learning.enforce_single_correct_option (011) contava as alternativas
-- corretas ANTES da mudança e recusava só se a contagem já fosse > 1. Num
-- UPDATE de false→true com uma alternativa já correta, a contagem "antes"
-- era 1 → 1 > 1 é falso → PASSAVA. E num INSERT, a contagem "antes" é a
-- mesma → PASSAVA. O QA mediu: um INSERT de uma alternativa 'E' com
-- is_correct=true deixou o item com DUAS corretas — e a publicação
-- (que exige exatamente uma) passou a recusar.
--
-- A CORREÇÃO: a regra passa a olhar o estado DEPOIS da mudança — as
-- alternativas corretas do item,excluding a linha que está mudando, mais
-- a linha nova. Vale para INSERT e para UPDATE.
--
-- E a letra derivada (022) passa a cobrir o INSERT: um constraint trigger
-- AFTER INSERT OR UPDATE, deferido para o COMMIT.

--
-- A CORREÇÃO:
--   · a comparação passa a ser com a TRILHA (que é por exame e por

CREATE OR REPLACE FUNCTION canone_ops.current_role() RETURNS text
LANGUAGE sql STABLE AS $$
  SELECT COALESCE(NULLIF(current_setting('role', true), ''), current_user)
$$;

COMMENT ON FUNCTION canone_ops.current_role() IS
  'Papel efetivo da requisição. O GUC "role" só muda por SET ROLE, que exige membro do papel de destino — não é forjável como um GUC de texto livre.';

CREATE OR REPLACE FUNCTION canone_ops.is_staff() RETURNS boolean
LANGUAGE sql STABLE AS $$
  SELECT ops.current_role() IN ('canone_staff', 'canone_admin')
$$;

COMMENT ON FUNCTION canone_ops.is_staff() IS
  'Staff é o PAPEL efetivo (canone_staff/canone_admin). NÃO é o GUC app.current_area: qualquer sessão escreve aquele GUC, e com ele o gabarito era revelado antes da tentativa.';

-- -----------------------------------------------------------------------------
-- 2 · EXATAMENTE UMA ALTERNATIVA CORRETA — ESTADO DEPOIS DA MUDANÇA
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_learning.enforce_single_correct_option() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  v_count int;
BEGIN
  IF NEW.is_correct THEN
    -- As alternativas corretas do item, EXCLUINDO a linha que muda (no
    -- INSERT ela ainda não existe; no UPDATE ela conta com o valor velho).
    SELECT count(*) INTO v_count
      FROM learning.item_option
     WHERE item_id = NEW.item_id AND is_correct AND id <> NEW.id;

    IF v_count + 1 > 1 THEN
      RAISE EXCEPTION 'Item % teria % alternativas corretas; o edital admite uma',
        NEW.item_id, v_count + 1 USING ERRCODE = '23514';
    END IF;
  END IF;
  RETURN NEW;
END $$;

DROP TRIGGER IF EXISTS trg_single_correct_option ON canone_learning.item_option;

CREATE TRIGGER trg_single_correct_option
  BEFORE INSERT OR UPDATE OF is_correct ON canone_learning.item_option
  FOR EACH ROW EXECUTE FUNCTION canone_learning.enforce_single_correct_option();

COMMENT ON FUNCTION canone_learning.enforce_single_correct_option() IS
  'Item de avaliação tem no máximo uma alternativa correta — contado no estado DEPOIS da mudança (INSERT incluído).';

-- 3 · CONFLITO DE NÚMERO DE QUESTÕES: CONTRA A TRILHA, SEMPRE
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION canone_ops.import_corpus_form(p_payload jsonb)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public, ops, editorial, taxonomy, learning, identity, norms
AS $$
DECLARE
  v_id         text;
  v_ano       int;
  v_edicao    int;
  v_tipo      text;
  v_exam_type text;
  v_banca     text;
  v_gab_type  text;
  v_questoes  jsonb;
  v_q         jsonb;
  v_ordinal   int;
  v_letter    text;
  v_annulled  boolean;
  v_total     int;
  v_ed_code   text;
  v_track_code text;
  v_specialty text;
  v_edition_id uuid;
  v_track_id   uuid;
  v_form_id    uuid;
  v_key_id     uuid;
  v_source_id  uuid;
  v_ct_id      uuid;
  v_ctv_id     uuid;
  v_item_id    uuid;
  v_stem      text;
  v_opts      jsonb;
  v_alt       jsonb;
  v_alt_key   text;
  v_alt_body  text;
  v_correct   text;
  v_content_id uuid;
  v_version_id uuid;
  v_existing   jsonb;
  v_count      int;
  v_conflict   int;
BEGIN
  -- 1 · O QUE CAIU DENTRO
  v_id         := p_payload->>'id';
  v_ano       := (p_payload->>'ano')::int;
  v_edicao    := (p_payload->>'edicao')::int;
  v_tipo      := upper(p_payload->>'tipo_prova');
  v_exam_type := p_payload->>'exam_type';
  v_banca     := p_payload->>'banca';
  v_gab_type  := lower(coalesce(p_payload->>'gabarito_type', 'preliminary'));
  v_questoes  := p_payload->'questoes';

  IF v_id IS NULL OR v_ano IS NULL OR v_edicao IS NULL OR v_tipo IS NULL
     OR v_exam_type IS NULL OR v_questoes IS NULL THEN
    RAISE EXCEPTION 'Acervo inválido: faltam id, ano, edicao, tipo_prova, exam_type ou questoes'
      USING ERRCODE = '22023';
  END IF;
  IF jsonb_typeof(v_questoes) <> 'array' THEN
    RAISE EXCEPTION 'Acervo %: questoes não é lista', v_id USING ERRCODE = '22023';
  END IF;
  IF v_gab_type NOT IN ('definitive', 'preliminary') THEN
    RAISE EXCEPTION 'Acervo %: gabarito_type % desconhecido', v_id, v_gab_type
      USING ERRCODE = '22023';
  END IF;
  -- O acervo fala 'definitive/preliminary'; o banco fala 'definitivo/preliminar'.
  -- O TRADUÇÃO É EXPLÍCITA — as palavras não são as mesmas em inglês e em
  -- português, e quem lê o banco tem que saber qual dos dois está vendo.
  v_gab_type := CASE v_gab_type WHEN 'definitive' THEN 'definitivo'
                                ELSE 'preliminar' END;

  v_total := jsonb_array_length(v_questoes);
  IF v_total = 0 THEN
    RAISE EXCEPTION 'Acervo %: nenhuma questão', v_id USING ERRCODE = '22023';
  END IF;

  -- O código da rodada segue o PADRÃO DO SEED ('2026.1', não '2026.01'):
  -- o mesmo exame tem que ter o mesmo nome em toda parte, senão a rodada
  -- vira duas rodadas e o acervo se parte ao meio.
  v_ed_code    := v_ano::text || '.' || v_edicao::text;
  v_track_code := CASE WHEN lower(v_exam_type) IN ('suf', 'suficiencia', 'suficiência')
                       THEN 'suficiencia' ELSE 'eqt' END;
  v_specialty := CASE WHEN v_track_code = 'eqt' THEN upper(v_exam_type) ELSE NULL END;

  -- 2 · A RODADA
  SELECT id INTO v_edition_id FROM taxonomy.exam_edition WHERE code = v_ed_code;
  IF NOT FOUND THEN
    INSERT INTO taxonomy.exam_edition
      (code, label, edition_number, year, application_date, question_count,
       passing_score, status, bank, exam_kind)
    VALUES (v_ed_code, 'Exame ' || v_ed_code, v_edicao, v_ano, NULL,
            CASE WHEN v_track_code = 'suficiencia' THEN v_total ELSE NULL END,
            NULL, 'realizada', v_banca,
            CASE WHEN v_track_code = 'suficiencia' THEN 'suficiencia' ELSE 'qualificacao_tecnica' END)
    RETURNING id INTO v_edition_id;
  ELSE
    -- A rodada tem como exame principal a Suficiência. Se a EQT chegou
    -- primeiro, o campo é corrigido agora — a regra é fixa (Suficiência é a
    -- principal), não é "o último que chegou vence".
    IF v_track_code = 'suficiencia' THEN
      UPDATE taxonomy.exam_edition SET exam_kind = 'suficiencia'
       WHERE id = v_edition_id AND exam_kind <> 'suficiencia';
    END IF;

  END IF;

  -- 3 · A TRILHA
  SELECT id INTO v_track_id FROM taxonomy.exam_track
   WHERE exam_edition_id = v_edition_id
     AND track_code = v_track_code
     AND COALESCE(specialty_code, '') = COALESCE(v_specialty, '');
  IF NOT FOUND THEN
    INSERT INTO taxonomy.exam_track
      (exam_edition_id, track_code, specialty_code, label, question_count, passing_score)
    VALUES (v_edition_id, v_track_code, v_specialty,
            CASE WHEN v_track_code = 'suficiencia' THEN 'Suficiência' ELSE 'EQT — ' || v_specialty END,
            v_total, NULL)
    RETURNING id INTO v_track_id;
  ELSE
    -- CONFLITO contra a TRILHA: a mesma trilha (mesmo exame, mesma
    -- especialidade) com outro número de questões é divergência de fonte, e
    -- divergência não se resolve sobrescrevendo. Vale para TODA variante,
    -- preliminar ou definitiva — o número é fato da PROVA, não do gabarito.
    SELECT question_count INTO v_count FROM taxonomy.exam_track WHERE id = v_track_id;
    IF v_count IS NOT NULL AND v_count <> v_total THEN
      RAISE EXCEPTION 'Conflito: a trilha % tem % questões no banco e % no acervo',
        v_track_code, v_count, v_total USING ERRCODE = '23505';
    END IF;
  END IF;

  -- A rodada da Suficiência recebe o número extraído do acervo: a prova É
  -- a fonte do número de questões dela. Sem isto, o conflito ficaria
  -- desligado para sempre (a rodada nasce NULL quando a primeira variante
  -- que chega é de EQT).
  IF v_track_code = 'suficiencia' THEN
    SELECT question_count INTO v_count FROM taxonomy.exam_edition WHERE id = v_edition_id;
    IF v_count IS NULL THEN
      UPDATE taxonomy.exam_edition SET question_count = v_total WHERE id = v_edition_id;
    ELSIF v_count <> v_total THEN
      RAISE EXCEPTION 'Conflito: a rodada % tem % questões no banco e % no acervo',
        v_ed_code, v_count, v_total USING ERRCODE = '23505';
    END IF;
  END IF;

  -- 4 · A VARIANTE
  SELECT id INTO v_form_id FROM learning.exam_form
   WHERE exam_track_id = v_track_id AND form_code = v_tipo;
  IF NOT FOUND THEN
    INSERT INTO learning.exam_form (exam_track_id, form_code, form_label)
    VALUES (v_track_id, v_tipo, 'Tipo ' || v_tipo)
    RETURNING id INTO v_form_id;
  END IF;

  -- 5 · A FONTE (o PDF da prova)
  SELECT id INTO v_source_id FROM editorial.source WHERE key = v_id;
  IF NOT FOUND THEN
    INSERT INTO editorial.source (key, kind, title, publisher, source_quote, consulted_at)
    VALUES (v_id, 'arquivo', 'Prova ' || v_ed_code || ' — ' || v_tipo, v_banca,
            'Prova oficial ' || v_ed_code || ', variante ' || v_tipo || ', ' || v_total || ' questões.',
            now())
    RETURNING id INTO v_source_id;
  END IF;

  -- 6 · AS QUESTÕES (payload canônico → projeção por trigger)
  SELECT ct.id, ctv.id INTO v_ct_id, v_ctv_id
    FROM editorial.content_type ct
    JOIN editorial.content_type_version ctv ON ctv.content_type_id = ct.id
   WHERE ct.key = 'questao' AND ctv.version = 1;
  IF v_ct_id IS NULL THEN
    RAISE EXCEPTION 'O tipo de conteúdo "questao" não existe' USING ERRCODE = '23503';
  END IF;

  FOR v_q IN SELECT a FROM jsonb_array_elements(v_questoes) AS a ORDER BY (a->>'numero')::int LOOP
    v_ordinal  := (v_q->>'numero')::int;
    v_letter   := v_q->>'gabarito';
    v_annulled := (v_letter IS NULL OR btrim(v_letter) = '' OR v_letter = '*');

    -- O ENUNCIADO: lido defensivamente, ausência é ERRO.
    v_stem := coalesce(
      v_q->>'enunciado', v_q->>'texto', v_q->>'questao', v_q->>'statement', v_q->>'pergunta'
    );
    IF v_stem IS NULL OR btrim(v_stem) = '' THEN
      RAISE EXCEPTION
        'Acervo %: a questão % não tem enunciado (nenhum dos campos enunciado|texto|questao|statement|pergunta existe)',
        v_id, v_ordinal USING ERRCODE = '22023';
    END IF;

    -- AS ALTERNATIVAS: objeto {A:..,B:..} ou lista [{ordinal,body}].
    v_opts := v_q->'alternativas';
    IF v_opts IS NULL THEN v_opts := v_q->'options'; END IF;
    IF v_opts IS NULL THEN v_opts := v_q->'opcoes'; END IF;
    IF v_opts IS NULL THEN
      RAISE EXCEPTION 'Acervo %: a questão % não tem alternativas', v_id, v_ordinal
        USING ERRCODE = '22023';
    END IF;

    v_alt := '[]'::jsonb;
    IF jsonb_typeof(v_opts) = 'object' THEN
      FOR v_alt_key IN SELECT jsonb_object_keys(v_opts) ORDER BY 1 LOOP
        v_alt_body := v_opts->>v_alt_key;
        v_alt := v_alt || jsonb_build_array(jsonb_build_object(
          'ordinal', v_alt_key, 'body', v_alt_body,
          'correct', (NOT v_annulled AND upper(v_alt_key) = upper(v_letter))));
      END LOOP;
    ELSIF jsonb_typeof(v_opts) = 'array' THEN
      FOR v_alt_key, v_alt_body IN
        SELECT coalesce(a->>'ordinal', a->>'letra'), coalesce(a->>'body', a->>'texto')
          FROM jsonb_array_elements(v_opts) a
      LOOP
        v_alt := v_alt || jsonb_build_array(jsonb_build_object(
          'ordinal', v_alt_key, 'body', v_alt_body,
          'correct', (NOT v_annulled AND upper(v_alt_key) = upper(v_letter))));
      END LOOP;
    ELSE
      RAISE EXCEPTION 'Acervo %: as alternativas da questão % não são lista nem objeto', v_id, v_ordinal
        USING ERRCODE = '22023';
    END IF;

    v_correct := CASE WHEN v_annulled THEN NULL ELSE upper(v_letter) END;

    -- O ITEM (idempotente pela chave do acervo)
    SELECT id INTO v_content_id FROM editorial.content_item WHERE key = v_id || '_q' || lpad(v_ordinal::text, 3, '0');
    IF NOT FOUND THEN
      INSERT INTO editorial.content_item
        (content_type_id, content_type_version_id, key, state, title)
      VALUES (v_ct_id, v_ctv_id, v_id || '_q' || lpad(v_ordinal::text, 3, '0'), 'rascunho', v_stem)
      RETURNING id INTO v_content_id;
    END IF;

    -- A VERSÃO: o payload é canônico; a projeção em learning.* é derivada.
    SELECT id INTO v_version_id FROM editorial.content_item_version
     WHERE content_item_id = v_content_id ORDER BY version DESC LIMIT 1;
    IF v_version_id IS NULL THEN
      INSERT INTO editorial.content_item_version (content_item_id, version, payload)
      VALUES (v_content_id, 1, jsonb_build_object(
                'kind', 'questao', 'stem', v_stem, 'options', v_alt))
      RETURNING id INTO v_version_id;
    ELSE
      UPDATE editorial.content_item_version
         SET payload = jsonb_build_object('kind','questao','stem',v_stem,'options',v_alt)
       WHERE id = v_version_id;
    END IF;
    UPDATE editorial.content_item SET current_version_id = v_version_id WHERE id = v_content_id;

    SELECT id INTO v_item_id FROM learning.item WHERE content_item_id = v_content_id;

    -- A POSIÇÃO NA PROVA
    INSERT INTO learning.exam_form_item (exam_form_id, ordinal, item_id)
    VALUES (v_form_id, v_ordinal, v_item_id)
    ON CONFLICT (exam_form_id, ordinal) DO UPDATE SET item_id = EXCLUDED.item_id;

  END LOOP;

  -- 7 · O GABARITO
  IF (SELECT count(*) FROM learning.exam_form_item WHERE exam_form_id = v_form_id) <> v_total THEN
    RAISE EXCEPTION 'Acervo %: a prova ficou com % questões de %',
      v_id, (SELECT count(*) FROM learning.exam_form_item WHERE exam_form_id = v_form_id), v_total
      USING ERRCODE = '23514';
  END IF;

  SELECT id INTO v_key_id FROM learning.answer_key
   WHERE exam_form_id = v_form_id AND status = v_gab_type;
  IF NOT FOUND THEN
    INSERT INTO learning.answer_key
      (exam_form_id, status, source_id, extraction)
    VALUES (v_form_id, v_gab_type, v_source_id,
            jsonb_build_object('method', 'importacao', 'acervo_id', v_id,
                               'parser_version', 'acervo-cfc-2.0'))
    RETURNING id INTO v_key_id;
  END IF;

  -- As entradas, na POSIÇÃO. A letra vem do acervo; a anulação é o
  -- gabarito ausente naquela posição.
  FOR v_q IN SELECT a FROM jsonb_array_elements(v_questoes) AS a ORDER BY (a->>'numero')::int LOOP
    v_ordinal  := (v_q->>'numero')::int;
    v_letter   := v_q->>'gabarito';
    v_annulled := (v_letter IS NULL OR btrim(v_letter) = '' OR v_letter = '*');
    INSERT INTO learning.answer_key_entry (answer_key_id, ordinal, correct_option, is_annulled)
    VALUES (v_key_id, v_ordinal, CASE WHEN v_annulled THEN NULL ELSE upper(v_letter) END, v_annulled)
    ON CONFLICT (answer_key_id, ordinal) DO UPDATE
      SET correct_option = EXCLUDED.correct_option,
          is_annulled = EXCLUDED.is_annulled;
  END LOOP;

  RETURN v_form_id;
END $$;
