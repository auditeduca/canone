CREATE OR REPLACE FUNCTION canone_ops.exam_start(p_exam_form_id uuid)
RETURNS uuid
LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path=pg_catalog AS $fn$
DECLARE v_uid uuid := auth.uid(); v_id uuid;
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'VERIFIED_JWT_REQUIRED' USING ERRCODE='42501'; END IF;
  -- Do not allow fake unpublished / unsealed official exams to be attempted.
  IF NOT EXISTS (
    SELECT 1 FROM canone_learning.exam_form f
      JOIN canone_taxonomy.exam_track t ON t.id=f.exam_track_id
      JOIN canone_taxonomy.exam_edition e ON e.id=t.exam_edition_id
      JOIN canone_learning.answer_key k ON k.exam_form_id=f.id AND k.status='definitivo'
    WHERE f.id=p_exam_form_id AND e.status='realizada'
  ) THEN RAISE EXCEPTION 'FORM_NOT_RELEASED_AND_SEALED' USING ERRCODE='23514'; END IF;
  INSERT INTO canone_learning.exam_attempt (user_id,exam_form_id)
    VALUES (v_uid,p_exam_form_id) RETURNING id INTO v_id;
  RETURN v_id;
END;
$fn$;

-- Never return the verdict or correct option from this method. The backend
-- must not serialize row counts (correct_count) while attempt is open.
CREATE OR REPLACE FUNCTION canone_ops.exam_answer(
 p_attempt_id uuid,p_ordinal integer,p_option text) RETURNS void
LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path=pg_catalog AS $fn$
DECLARE
 v_uid uuid := auth.uid(); v_attempt canone_learning.exam_attempt%ROWTYPE;
 v_item uuid; v_expected text; v_annulled boolean; v_verdict text;
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'VERIFIED_JWT_REQUIRED' USING ERRCODE='42501'; END IF;
  IF p_ordinal IS NULL OR p_ordinal < 1 OR p_option IS NULL OR
     upper(p_option) NOT IN ('A','B','C','D','E') OR length(p_option)<>1 THEN
    RAISE EXCEPTION 'INVALID_ANSWER' USING ERRCODE='23514'; END IF;
  -- Lock serializes repeat submissions against finish and against each other.
  SELECT * INTO v_attempt FROM canone_learning.exam_attempt
    WHERE id=p_attempt_id FOR UPDATE;
  IF NOT FOUND THEN RAISE EXCEPTION 'ATTEMPT_NOT_FOUND' USING ERRCODE='23503'; END IF;
  IF v_attempt.user_id<>v_uid THEN RAISE EXCEPTION 'NOT_OWNER' USING ERRCODE='42501'; END IF;
  IF v_attempt.status<>'em_andamento' THEN
    RAISE EXCEPTION 'ATTEMPT_CLOSED' USING ERRCODE='23514'; END IF;
  IF v_attempt.exam_form_id IS NULL THEN
    RAISE EXCEPTION 'MOCK_EXAM_NOT_IN_THIS_PILOT' USING ERRCODE='23514'; END IF;
  SELECT f.item_id,ent.correct_option,ent.is_annulled
    INTO v_item,v_expected,v_annulled
    FROM canone_learning.exam_form_item f
    JOIN canone_learning.answer_key k ON k.exam_form_id=f.exam_form_id AND k.status='definitivo'
    JOIN canone_learning.answer_key_entry ent ON ent.answer_key_id=k.id AND ent.ordinal=f.ordinal
    WHERE f.exam_form_id=v_attempt.exam_form_id AND f.ordinal=p_ordinal;
  IF NOT FOUND THEN RAISE EXCEPTION 'FORM_KEY_POSITION_MISSING' USING ERRCODE='23514'; END IF;
  v_verdict := CASE WHEN v_annulled THEN 'anulada'
                    WHEN upper(p_option)=v_expected THEN 'correta' ELSE 'errada' END;
  INSERT INTO canone_learning.exam_attempt_answer
    (attempt_id,ordinal,item_id,chosen_option,verdict)
    VALUES (p_attempt_id,p_ordinal,v_item,upper(p_option),v_verdict)
    ON CONFLICT (attempt_id,ordinal) DO UPDATE
      SET chosen_option=EXCLUDED.chosen_option,verdict=EXCLUDED.verdict,answered_at=clock_timestamp();
  -- Recalculate, never increment (repeat submission cannot inflate counts).
  UPDATE canone_learning.exam_attempt a
   SET answered_count=s.total,correct_count=s.right_count,
       wrong_count=s.wrong_count,annulled_count=s.annul_count
  FROM (SELECT count(*)::int total,
            (count(*) FILTER (WHERE verdict='correta'))::int right_count,
            (count(*) FILTER (WHERE verdict='errada'))::int wrong_count,
            (count(*) FILTER (WHERE verdict='anulada'))::int annul_count
        FROM canone_learning.exam_attempt_answer WHERE attempt_id=p_attempt_id) s
  WHERE a.id=p_attempt_id;
END;
$fn$;

CREATE OR REPLACE FUNCTION canone_ops.exam_finish(p_attempt_id uuid)
RETURNS TABLE(attempt_id uuid,answered_count integer,correct_count integer,
              wrong_count integer,annulled_count integer)
LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path=pg_catalog AS $fn$
DECLARE v_uid uuid:=auth.uid(); v_attempt canone_learning.exam_attempt%ROWTYPE;
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'VERIFIED_JWT_REQUIRED' USING ERRCODE='42501'; END IF;
  SELECT * INTO v_attempt FROM canone_learning.exam_attempt WHERE id=p_attempt_id FOR UPDATE;
  IF NOT FOUND THEN RAISE EXCEPTION 'ATTEMPT_NOT_FOUND' USING ERRCODE='23503'; END IF;
  IF v_attempt.user_id<>v_uid THEN RAISE EXCEPTION 'NOT_OWNER' USING ERRCODE='42501'; END IF;
  IF v_attempt.status='abandonada' THEN
    RAISE EXCEPTION 'ATTEMPT_ABANDONED' USING ERRCODE='23514'; END IF;
  IF v_attempt.status='em_andamento' THEN
    IF v_attempt.answered_count=0 THEN
      RAISE EXCEPTION 'NO_ANSWERS' USING ERRCODE='23514'; END IF;
    UPDATE canone_learning.exam_attempt a
       SET status='concluida',finished_at=clock_timestamp()
     WHERE a.id=p_attempt_id RETURNING * INTO v_attempt;
  END IF;
  -- No approval or official CFC cut score invented here.
  RETURN QUERY SELECT v_attempt.id,v_attempt.answered_count,
                 v_attempt.correct_count,v_attempt.wrong_count,v_attempt.annulled_count;
END;
$fn$;

CREATE OR REPLACE FUNCTION canone_ops.exam_reveal(p_attempt_id uuid)
RETURNS TABLE(ordinal integer,chosen_option text,verdict text,
              correct_option text,is_annulled boolean)
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=pg_catalog AS $fn$
DECLARE v_uid uuid:=auth.uid(); v_attempt canone_learning.exam_attempt%ROWTYPE;
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'VERIFIED_JWT_REQUIRED' USING ERRCODE='42501'; END IF;
  SELECT * INTO v_attempt FROM canone_learning.exam_attempt WHERE id=p_attempt_id;
  IF NOT FOUND THEN RAISE EXCEPTION 'ATTEMPT_NOT_FOUND' USING ERRCODE='23503'; END IF;
  IF v_attempt.user_id<>v_uid THEN RAISE EXCEPTION 'NOT_OWNER' USING ERRCODE='42501'; END IF;
  IF v_attempt.status<>'concluida' THEN
    RAISE EXCEPTION 'KEY_ONLY_AFTER_FINISH' USING ERRCODE='42501'; END IF;
  RETURN QUERY
    SELECT a.ordinal,a.chosen_option,a.verdict,e.correct_option,e.is_annulled
    FROM canone_learning.exam_attempt_answer a
    JOIN canone_learning.answer_key k ON k.exam_form_id=v_attempt.exam_form_id AND k.status='definitivo'
    JOIN canone_learning.answer_key_entry e ON e.answer_key_id=k.id AND e.ordinal=a.ordinal
    WHERE a.attempt_id=p_attempt_id ORDER BY a.ordinal;
END;
$fn$;

-- Only a completed, server-graded answer can become a system-based SRS event.
-- Retries return the same event id; an annulled item never becomes an error.
CREATE OR REPLACE FUNCTION canone_ops.exam_to_srs(p_attempt_id uuid,p_ordinal integer)
RETURNS uuid
LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path=pg_catalog AS $fn$
DECLARE
 v_uid uuid:=auth.uid();v_attempt canone_learning.exam_attempt%ROWTYPE;
 v_event uuid;v_item uuid;v_verdict text;v_correct boolean;
 v_grade canone_learning.review_grade;
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'VERIFIED_JWT_REQUIRED' USING ERRCODE='42501'; END IF;
  SELECT * INTO v_attempt FROM canone_learning.exam_attempt WHERE id=p_attempt_id FOR UPDATE;
  IF NOT FOUND THEN RAISE EXCEPTION 'ATTEMPT_NOT_FOUND' USING ERRCODE='23503'; END IF;
  IF v_attempt.user_id<>v_uid THEN RAISE EXCEPTION 'NOT_OWNER' USING ERRCODE='42501'; END IF;
  IF v_attempt.status<>'concluida' THEN RAISE EXCEPTION 'FINISH_BEFORE_SRS' USING ERRCODE='23514'; END IF;
  SELECT r.review_event_id INTO v_event FROM canone_learning.exam_review_receipt r
    WHERE r.attempt_id=p_attempt_id AND r.ordinal=p_ordinal;
  IF FOUND THEN RETURN v_event; END IF;
  SELECT a.item_id,a.verdict INTO v_item,v_verdict
    FROM canone_learning.exam_attempt_answer a
    WHERE a.attempt_id=p_attempt_id AND a.ordinal=p_ordinal;
  IF NOT FOUND THEN RAISE EXCEPTION 'ANSWER_NOT_FOUND' USING ERRCODE='23503'; END IF;
  IF v_verdict='anulada' THEN RETURN NULL; END IF;
  v_correct := v_verdict='correta';
  v_grade := CASE WHEN v_correct THEN 'facil'::canone_learning.review_grade
                  ELSE 'errei'::canone_learning.review_grade END;
  INSERT INTO canone_learning.review_event
    (user_id,item_id,grade,basis,correct,algorithm_version)
    VALUES (v_uid,v_item,v_grade,'sistema',v_correct,'canone-srs-exam-v1')
    RETURNING id INTO v_event;
  INSERT INTO canone_learning.user_item_state
    (user_id,item_id,state,stability_days,difficulty,due_at,last_review_at,
     last_grade,last_interval_days,repetitions,lapses,algorithm_version)
  VALUES (v_uid,v_item,CASE WHEN v_correct THEN 'aprendendo' ELSE 'reaprendendo' END,
          CASE WHEN v_correct THEN 1.5 ELSE 0.4 END,
          CASE WHEN v_correct THEN 4.65 ELSE 5.85 END,
          clock_timestamp()+make_interval(days=>CASE WHEN v_correct THEN 2 ELSE 0 END),
          clock_timestamp(),v_grade,CASE WHEN v_correct THEN 2 ELSE 0 END,
          1,CASE WHEN v_correct THEN 0 ELSE 1 END,'canone-srs-exam-v1')
  ON CONFLICT (user_id,item_id) DO UPDATE SET
    state=CASE WHEN v_correct THEN 'revisao' ELSE 'reaprendendo' END,
    stability_days=CASE WHEN v_correct THEN
      least(365.0,canone_learning.user_item_state.stability_days*2.4+1.5)
      ELSE greatest(0.4,canone_learning.user_item_state.stability_days*0.35) END,
    difficulty=CASE WHEN v_correct THEN
      greatest(1.0,canone_learning.user_item_state.difficulty-0.35)
      ELSE least(10.0,canone_learning.user_item_state.difficulty+0.85) END,
    due_at=clock_timestamp()+make_interval(days=>CASE WHEN v_correct THEN 2 ELSE 0 END),
    last_review_at=clock_timestamp(),last_grade=v_grade,
    last_interval_days=CASE WHEN v_correct THEN 2 ELSE 0 END,
    repetitions=canone_learning.user_item_state.repetitions+1,
    lapses=canone_learning.user_item_state.lapses+CASE WHEN v_correct THEN 0 ELSE 1 END,
    fragile_streak=0,algorithm_version='canone-srs-exam-v1';
  INSERT INTO canone_learning.exam_review_receipt(attempt_id,ordinal,review_event_id,user_id)
    VALUES (p_attempt_id,p_ordinal,v_event,v_uid);
  RETURN v_event;
END;
$fn$;
