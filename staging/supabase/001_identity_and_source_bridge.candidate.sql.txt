-- Source: S22 (001/002) projected through the 0.23.0-alpha.2 crosswalk.
-- Target: a NEW, dedicated Supabase project. Never apply to Calculadoras-Smart.
-- This is a deliberately bounded identity/provenance pilot, not all 30 migrations.
-- Do not promote until independently verified in ephemeral PostgreSQL and staging.
BEGIN;
SET LOCAL search_path = pg_catalog;

-- Defense against running in the shared source project that already carries
-- the Cânone governance catalog. This is NOT a universal environment attestation.
DO $canone_environment_guard$
BEGIN
  IF to_regclass('canone_control.migration_catalog') IS NOT NULL THEN
    RAISE EXCEPTION 'CANONE_REFUSE_SHARED_SOURCE: existing canone_control.migration_catalog';
  END IF;
  IF to_regclass('auth.users') IS NULL THEN
    RAISE EXCEPTION 'CANONE_REQUIRE_SUPABASE_AUTH: auth.users is missing';
  END IF;
END;
$canone_environment_guard$;

CREATE SCHEMA canone_identity;
CREATE SCHEMA canone_bridge;
REVOKE ALL ON SCHEMA canone_identity FROM PUBLIC;
REVOKE ALL ON SCHEMA canone_bridge FROM PUBLIC;

-- Supabase Auth is the only credential, password, OTP and login-session store.
-- No identity.account, identity.session or identity.magic_link replica is created.
CREATE TABLE canone_identity.user_profile (
    user_id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    display_name text NOT NULL CHECK (length(btrim(display_name)) BETWEEN 1 AND 120),
    given_name text,
    family_name text,
    timezone text NOT NULL DEFAULT 'America/Sao_Paulo',
    locale text NOT NULL DEFAULT 'pt-BR',
    avatar_url text,
    onboarded_at timestamptz,
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE canone_identity.private_profile (
    user_id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    birth_date date,
    cpf_encrypted bytea,
    cpf_last4 char(4),
    phone_e164 text,
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE canone_identity.role (
    key text PRIMARY KEY,
    label text NOT NULL,
    scope text NOT NULL CHECK (scope IN ('publica','assinante','staff','admin')),
    requires_subscription boolean NOT NULL DEFAULT false,
    description text
);
CREATE TABLE canone_identity.user_role (
    user_id uuid NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    role_key text NOT NULL REFERENCES canone_identity.role(key),
    granted_by uuid REFERENCES auth.users(id),
    granted_at timestamptz NOT NULL DEFAULT now(),
    valid_until timestamptz,
    revoked_at timestamptz,
    reason text,
    PRIMARY KEY (user_id, role_key, granted_at)
);
CREATE INDEX ix_canone_user_role_active ON canone_identity.user_role(user_id)
    WHERE revoked_at IS NULL;
CREATE TABLE canone_identity.consent_purpose (
    key text PRIMARY KEY,
    label text NOT NULL,
    legal_basis text NOT NULL,
    version integer NOT NULL DEFAULT 1 CHECK (version > 0),
    active boolean NOT NULL DEFAULT true,
    required boolean NOT NULL DEFAULT false
);
CREATE TABLE canone_identity.consent (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id uuid NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    purpose_key text NOT NULL REFERENCES canone_identity.consent_purpose(key),
    purpose_version integer NOT NULL CHECK (purpose_version > 0),
    granted boolean NOT NULL,
    legal_basis text NOT NULL,
    evidence jsonb NOT NULL DEFAULT '{}'::jsonb,
    collected_at timestamptz NOT NULL DEFAULT now(),
    collected_by text NOT NULL DEFAULT 'server'
);
CREATE INDEX ix_canone_consent_user ON canone_identity.consent(user_id, purpose_key, collected_at DESC);

-- Preserve ownership as metadata only; do not import restricted corpus bytes.
CREATE TABLE canone_bridge.source_binding (
    id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    source_system text NOT NULL,
    source_kind text NOT NULL,
    external_object_id text NOT NULL,
    source_version text NOT NULL,
    source_sha256 text CHECK (source_sha256 IS NULL OR source_sha256 ~ '^[0-9a-f]{64}$'),
    rights_status text NOT NULL DEFAULT 'unverified'
        CHECK (rights_status IN ('unverified','permitted','restricted','rejected')),
    evidence_status text NOT NULL DEFAULT 'unverified'
        CHECK (evidence_status IN ('unverified','verified','revoked')),
    created_at timestamptz NOT NULL DEFAULT now(),
    UNIQUE(source_system, source_kind, external_object_id, source_version)
);

-- All tables are RLS-enabled and forced, including private audit/role sources.
ALTER TABLE canone_identity.user_profile ENABLE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.user_profile FORCE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.private_profile ENABLE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.private_profile FORCE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.role ENABLE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.role FORCE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.user_role ENABLE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.user_role FORCE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.consent_purpose ENABLE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.consent_purpose FORCE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.consent ENABLE ROW LEVEL SECURITY;
ALTER TABLE canone_identity.consent FORCE ROW LEVEL SECURITY;
ALTER TABLE canone_bridge.source_binding ENABLE ROW LEVEL SECURITY;
ALTER TABLE canone_bridge.source_binding FORCE ROW LEVEL SECURITY;

-- Only explicitly exposed, non-sensitive profile columns are user-editable.
GRANT USAGE ON SCHEMA canone_identity TO authenticated;
GRANT SELECT (user_id, display_name, given_name, family_name, timezone, locale,
              avatar_url, onboarded_at, created_at, updated_at)
    ON canone_identity.user_profile TO authenticated;
GRANT INSERT (user_id, display_name, given_name, family_name, timezone, locale, avatar_url)
    ON canone_identity.user_profile TO authenticated;
GRANT UPDATE (display_name, given_name, family_name, timezone, locale, avatar_url)
    ON canone_identity.user_profile TO authenticated;

CREATE POLICY canone_profile_self_select ON canone_identity.user_profile
    FOR SELECT TO authenticated USING ((SELECT auth.uid()) = user_id);
CREATE POLICY canone_profile_self_insert ON canone_identity.user_profile
    FOR INSERT TO authenticated WITH CHECK ((SELECT auth.uid()) = user_id);
CREATE POLICY canone_profile_self_update ON canone_identity.user_profile
    FOR UPDATE TO authenticated
    USING ((SELECT auth.uid()) = user_id)
    WITH CHECK ((SELECT auth.uid()) = user_id);

-- No direct client grants or policies on private_profile, user_role, role,
-- consent, consent_purpose or source_binding. Administrative writes are a
-- future separately approved server contract, not a client-writable shortcut.
COMMIT;
